JAVASCRIPT FOR SHOW PASSWORD

onclick=" show_password()"
password id = 'get_password"
icon id = 'get_eye'

function show_password() {

     const get_password = document.getElementById("get_password");
     const get_eye = document.getElementById("get_eye");
     
     if(get_password.type === 'password'){
          get_password.type = 'test';
          get_eye.classList.replace("fa_eye_slash", "fa_eye");
} else {
     
       get_password.type = 'password';
       get_eye.classList.replace("fa_eye", "fa_eye_slash")

}
